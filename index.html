function updateHolidayCountdown() { // 1. 將今天的時間統一歸零至午夜 00:00:00 const today = new Date(); today.setHours(0, 0, 0, 0); // 2. 正確設置香港公眾假期（注意：JS 月份為 0 至 11，9月為 8，10月為 9） const holidays = [ { name: "中秋節後翌日", date: new Date(2026, 8, 26) }, { name: "國慶日", date: new Date(2026, 9, 1) }, { name: "重陽節翌日", date: new Date(2026, 9, 19) }, { name: "聖誕節", date: new Date(2026, 11, 25) }, { name: "聖誕節後第一個周日", date: new Date(2026, 11, 26) }, { name: "元旦", date: new Date(2027, 0, 1) }, { name: "農曆年初一", date: new Date(2027, 1, 6) } ]; // 3. 尋找今天或即將到來的假期 const nextHoliday = holidays.find(h => h.date >= today); if (nextHoliday) { const diffTime = nextHoliday.date - today; const diffDays = Math.round(diffTime / (1000 * 60 * 60 * 24)); const text = (diffDays === 0) ? `今日係 ${nextHoliday.name} 假期！` : `距離 ${nextHoliday.name}：仲有 ${diffDays} 天`; document.getElementById('countdown').textContent = text; } }
